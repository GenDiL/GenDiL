cmake_minimum_required(VERSION 3.25.2)
project(GenDiLInstalledConsumer LANGUAGES C CXX)

if(GENDIL_TEST_BACKEND STREQUAL "CUDA")
  enable_language(CUDA)
elseif(GENDIL_TEST_BACKEND STREQUAL "HIP")
  enable_language(HIP)
elseif(NOT GENDIL_TEST_BACKEND STREQUAL "CPU")
  message(FATAL_ERROR "GENDIL_TEST_BACKEND must be CPU, CUDA, or HIP")
endif()

set(consumer_install_prefix "${CMAKE_INSTALL_PREFIX}")
find_package(GENDIL CONFIG REQUIRED)
if(NOT CMAKE_INSTALL_PREFIX STREQUAL consumer_install_prefix)
  message(FATAL_ERROR "Finding GenDiL changed the consumer's CMAKE_INSTALL_PREFIX")
endif()

file(REAL_PATH "${GENDIL_EXPECTED_INCLUDE_DIR}" expected_include)
file(REAL_PATH "${GENDIL_INCLUDE_DIRS}" actual_include)
if(NOT actual_include STREQUAL expected_include OR
   NOT EXISTS "${GENDIL_INCLUDE_DIRS}/gendil/gendil.hpp")
  message(FATAL_ERROR "GENDIL_INCLUDE_DIRS does not identify the relocated installation")
endif()
cmake_path(IS_PREFIX GENDIL_EXPECTED_PREFIX "${GENDIL_DIR}" NORMALIZE found_relocated)
if(NOT found_relocated)
  message(FATAL_ERROR "The consumer found GenDiL outside its relocated installation")
endif()

get_target_property(target_includes GENDIL::GENDIL INTERFACE_INCLUDE_DIRECTORIES)
if("${GENDIL_PRODUCER_SOURCE}/include" IN_LIST target_includes)
  message(FATAL_ERROR "The installed target exposes GenDiL's source-tree headers")
endif()
if(GENDIL_TEST_BACKEND STREQUAL "CUDA" AND NOT GENDIL_USE_CUDA)
  message(FATAL_ERROR "The installed package lost its CUDA configuration")
elseif(GENDIL_TEST_BACKEND STREQUAL "HIP" AND NOT GENDIL_USE_HIP)
  message(FATAL_ERROR "The installed package lost its HIP configuration")
endif()

add_executable(installed-consumer main.cpp)
if(NOT GENDIL_TEST_BACKEND STREQUAL "CPU")
  set_source_files_properties(main.cpp PROPERTIES LANGUAGE "${GENDIL_TEST_BACKEND}")
endif()
target_link_libraries(installed-consumer PRIVATE GENDIL::GENDIL)

enable_testing()
add_test(NAME installed-consumer COMMAND installed-consumer)
